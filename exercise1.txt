using System;

namespace StudentAndStoreApp
{
    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("         C# CONSOLE APPLICATION EXERCISES         ");

            // ---------------------------------------------------------
            // 1. STUDENT INFORMATION
            // ---------------------------------------------------------
            Console.WriteLine("--- 1. STUDENT INFORMATION ---");
            Console.Write("Enter your Name: ");
            string name = Console.ReadLine();

            Console.Write("Enter your Course: ");
            string course = Console.ReadLine();

            Console.Write("Enter your Year Level: ");
            string yearLevel = Console.ReadLine();

            Console.Write("Enter your Section: ");
            string section = Console.ReadLine();

            Console.WriteLine("\n[Student Details]");
            Console.WriteLine("Name: " + name);
            Console.WriteLine("Course: " + course);
            Console.WriteLine("Year Level: " + yearLevel);
            Console.WriteLine("Section: " + section);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 2. BASIC CALCULATOR

            Console.WriteLine("--- 2. BASIC CALCULATOR ---");
            Console.Write("Enter first number: ");
            double num1 = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter second number: ");
            double num2 = Convert.ToDouble(Console.ReadLine());

            Console.WriteLine("Addition (+): " + (num1 + num2));
            Console.WriteLine("Subtraction (-): " + (num1 - num2));
            Console.WriteLine("Multiplication (*): " + (num1 * num2));

            if (num2 != 0)
            {
                Console.WriteLine("Division (/): " + (num1 / num2));
            }
            else
            {
                Console.WriteLine("Division (/): Cannot divide by zero.");
            }
            Console.WriteLine();

            // ---------------------------------------------------------
            // 3. AREA OF A RECTANGLE

            Console.WriteLine("--- 3. AREA OF A RECTANGLE ---");
            Console.Write("Enter Length: ");
            double length = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Width: ");
            double width = Convert.ToDouble(Console.ReadLine());

            double area = length * width;
            Console.WriteLine("Area of Rectangle: " + area);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 4. STUDENT GRADE
            // ---------------------------------------------------------
            Console.WriteLine("--- 4. STUDENT GRADE ---");
            Console.Write("Enter Prelim Grade: ");
            double prelim = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Midterm Grade: ");
            double midterm = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Final Grade: ");
            double finalGrade = Convert.ToDouble(Console.ReadLine());

            double gradeAverage = (prelim + midterm + finalGrade) / 3;
            Console.WriteLine("Average Grade: " + gradeAverage);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 5. BOOLEAN INPUT
            // ---------------------------------------------------------
            Console.WriteLine("--- 5. BOOLEAN INPUT ---");
            Console.Write("Are you a student? (true/false): ");
            bool isStudent = Convert.ToBoolean(Console.ReadLine());
            Console.WriteLine("Is Student: " + isStudent);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 6. AGE VERIFICATION
            // ---------------------------------------------------------
            Console.WriteLine("--- 6. AGE VERIFICATION ---");
            Console.Write("Enter your Age: ");
            int age = Convert.ToInt32(Console.ReadLine());

            bool isAdult = age >= 18;
            Console.WriteLine("Is 18 or older (isAdult): " + isAdult);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 7. FULL NAME
            // ---------------------------------------------------------
            Console.WriteLine("--- 7. FULL NAME ---");
            Console.Write("Enter First Name: ");
            string firstName = Console.ReadLine();

            Console.Write("Enter Last Name: ");
            string lastName = Console.ReadLine();

            string fullName = firstName + " " + lastName;
            Console.WriteLine("Full Name: " + fullName);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 8. SHOPPING CALCULATOR
            // ---------------------------------------------------------
            Console.WriteLine("--- 8. SHOPPING CALCULATOR ---");
            Console.Write("Enter Product Name: ");
            string productName = Console.ReadLine();

            Console.Write("Enter Product Price: ");
            double productPrice = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Quantity: ");
            int productQuantity = Convert.ToInt32(Console.ReadLine());

            double shoppingTotal = productPrice * productQuantity;
            Console.WriteLine("Total Cost for " + productName + ": " + shoppingTotal);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 9. SALARY CALCULATOR
            // ---------------------------------------------------------
            Console.WriteLine("--- 9. SALARY CALCULATOR ---");
            Console.Write("Enter Employee Name: ");
            string employeeName = Console.ReadLine();

            Console.Write("Enter Hours Worked: ");
            double hoursWorked = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Hourly Rate: ");
            double hourlyRate = Convert.ToDouble(Console.ReadLine());

            double salary = hoursWorked * hourlyRate;
            Console.WriteLine("Salary of " + employeeName + ": " + salary);
            Console.WriteLine();

            // ---------------------------------------------------------
            // 10. MINI STUDENT INFORMATION SYSTEM
            // ---------------------------------------------------------
            Console.WriteLine("--- 10. MINI STUDENT INFORMATION SYSTEM ---");
            Console.Write("Enter Name: ");
            string sysName = Console.ReadLine();

            Console.Write("Enter Course: ");
            string sysCourse = Console.ReadLine();

            Console.Write("Enter Age: ");
            int sysAge = Convert.ToInt32(Console.ReadLine());

            Console.Write("Enter Year Level: ");
            string sysYearLevel = Console.ReadLine();

            Console.Write("Enter Grade 1: ");
            double g1 = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Grade 2: ");
            double g2 = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Grade 3: ");
            double g3 = Convert.ToDouble(Console.ReadLine());

            double sysAverage = (g1 + g2 + g3) / 3;
            bool sysIsAdult = sysAge >= 18;

            Console.WriteLine("\n[MINI SIS RECORD]");
            Console.WriteLine("Name: " + sysName);
            Console.WriteLine("Course: " + sysCourse);
            Console.WriteLine("Age: " + sysAge);
            Console.WriteLine("Year Level: " + sysYearLevel);
            Console.WriteLine("Average Grade: " + sysAverage);
            Console.WriteLine("Is 18 or older: " + sysIsAdult);
            Console.WriteLine();

            // ---------------------------------------------------------
            // FINAL CHALLENGE: SIMPLE STORE RECEIPT
            // ---------------------------------------------------------
            Console.WriteLine("==================================================");
            Console.WriteLine("      FINAL CHALLENGE: SIMPLE STORE RECEIPT       ");
            Console.WriteLine("==================================================");
            Console.Write("Enter Customer Name: ");
            string customerName = Console.ReadLine();

            // Product 1 Input
            Console.WriteLine("\n[ Product 1 Details ]");
            Console.Write("Enter Product Name: ");
            string prod1Name = Console.ReadLine();

            Console.Write("Enter Product Price: ");
            double prod1Price = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Quantity: ");
            int prod1Qty = Convert.ToInt32(Console.ReadLine());

            // Product 2 Input
            Console.WriteLine("\n[ Product 2 Details ]");
            Console.Write("Enter Product Name: ");
            string prod2Name = Console.ReadLine();

            Console.Write("Enter Product Price: ");
            double prod2Price = Convert.ToDouble(Console.ReadLine());

            Console.Write("Enter Quantity: ");
            int prod2Qty = Convert.ToInt32(Console.ReadLine());

            // Subtotals & Total Calculation
            double subtotal1 = prod1Price * prod1Qty;
            double subtotal2 = prod2Price * prod2Qty;
            double grandTotal = subtotal1 + subtotal2;

            // Display Receipt
            Console.WriteLine("\n==================================================");
            Console.WriteLine("                  STORE RECEIPT                   ");
            Console.WriteLine("==================================================");
            Console.WriteLine("Customer Name: " + customerName);
            Console.WriteLine("--------------------------------------------------");
            Console.WriteLine("Product 1: " + prod1Name);
            Console.WriteLine("Price: " + prod1Price + " | Qty: " + prod1Qty + " | Subtotal: " + subtotal1);
            Console.WriteLine("--------------------------------------------------");
            Console.WriteLine("Product 2: " + prod2Name);
            Console.WriteLine("Price: " + prod2Price + " | Qty: " + prod2Qty + " | Subtotal: " + subtotal2);
            Console.WriteLine("--------------------------------------------------");
            Console.WriteLine("GRAND TOTAL: " + grandTotal);
            Console.WriteLine("==================================================");

            Console.WriteLine("\nProgram completed. Press any key to exit.");
            Console.ReadKey();
        }
    }
}
